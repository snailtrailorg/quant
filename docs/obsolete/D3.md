from unittest.mock import patch

        r = analyze_convertible_terms(fake_terms)
    assert "测试转债" in r["summary"]

# 2. pull_cb_basic（mock get_pro）
def test_pull_cb_basic():
    import pandas as pd
    fake_df = pd.DataFrame([fake_terms])
    with patch("src.data_platform.adapters.tushare_adapter.get_pro") as m_pro:
        m_pro.return_value.cb_basic.return_value = fake_df
        from src.data_platform.adapters.tushare_adapter import pull_cb_basic
        r = pull_cb_basic("113549.SH")
    assert r["bond_short_name"] == "测试转债"
