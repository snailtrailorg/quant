  ```python
      #        get_channel("wechat_work").send(

      #            body=f"{budget.provider or '全局'} 今日 {today_tokens} token 超阈值 {阈值}（{alert_threshold_pct}%）",
      #            level="warn")
      #      alerts.append({provider, today_tokens, limit, threshold, pct})
      # 3. 返回 {"checked": N, "alerts": alerts}
  ```

```python
from unittest.mock import patch, MagicMock
from fastapi.testclient import TestClient
from src.web_api.main import app
client = TestClient(app)

def _mock_conn_budget(rows_budget, usage_total=5000):
    """mock get_conn：llm_budget 查 + llm_usage 聚合"""
    mock_conn = MagicMock()
    mock_conn.__enter__.return_value = mock_conn
    # budget 行
    mock_conn.execute.return_value.fetchall.return_value = rows_budget
    # usage 聚合（scalar）-- check_budget_alerts 内第二次 execute
    # 实现时按实际 execute 调用顺序 mock；此处示意
    return mock_conn

# 1. 列预算（mock get_conn）
def test_list_budget():
    mock_conn = MagicMock(); mock_conn.__enter__.return_value = mock_conn
    cur = MagicMock(); cur.fetchall.return_value = [(1, None, 100000, None, 80, True)]
    mock_conn.execute.return_value = cur
    with patch("src.web_api.main.get_conn", return_value=mock_conn), \
         patch("src.web_api.main.require_role") as m:
        m.return_value = lambda *a, **k: {"sub":"1","username":"admin","role":"admin"}
        r = client.get("/api/llm-budget")
    assert r.status_code == 200

# 2. 超阈值告警（mock get_conn + get_channel）
def test_check_alert_triggers():
    # budget: limit=10000, threshold=80 -> 阈值 8000; usage_today=9000 -> 超阈值
    mock_conn = MagicMock(); mock_conn.__enter__.return_value = mock_conn
    cur_budget = MagicMock(); cur_budget.fetchall.return_value = [(1, None, 10000, None, 80, True)]
    cur_usage = MagicMock(); cur_usage.fetchone.return_value = (9000,)
    mock_conn.execute.side_effect = [cur_budget, cur_usage]
    mock_channel = MagicMock()
    with patch("src.web_api.main.get_conn", return_value=mock_conn), \
         patch("src.alert_notify.channel.get_channel", return_value=mock_channel), \
         patch("src.web_api.main.require_role") as m:
        m.return_value = lambda *a, **k: {"sub":"1","username":"admin","role":"admin"}
        r = client.post("/api/llm-budget/check")
    assert r.status_code == 200
    assert r.json()["checked"] >= 1
    assert len(r.json()["alerts"]) >= 1
    mock_channel.send.assert_called_once()  # 发了告警

# 3. 未超阈值不发告警
def test_check_no_alert():
    # limit=10000, threshold=80 -> 8000; usage=5000 -> 未超
    mock_conn = MagicMock(); mock_conn.__enter__.return_value = mock_conn
    cur_budget = MagicMock(); cur_budget.fetchall.return_value = [(1, None, 10000, None, 80, True)]
    cur_usage = MagicMock(); cur_usage.fetchone.return_value = (5000,)
    mock_conn.execute.side_effect = [cur_budget, cur_usage]
    mock_channel = MagicMock()
    with patch("src.web_api.main.get_conn", return_value=mock_conn), \
         patch("src.alert_notify.channel.get_channel", return_value=mock_channel), \
         patch("src.web_api.main.require_role") as m:
        m.return_value = lambda *a, **k: {"sub":"1","username":"admin","role":"admin"}
        r = client.post("/api/llm-budget/check")
    assert r.status_code == 200
    assert len(r.json()["alerts"]) == 0
    mock_channel.send.assert_not_called()
```
> ⚠️ mock require_role / get_conn 的 patch 路径以实际端点 Depends + 调用为准。Codex 实现时按 conftest + 实际机制调整，**核心断言**：超阈值时 `get_channel().send` 被调 + alerts 非空；未超时 send 不调。
