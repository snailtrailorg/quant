## 分发引擎契约（dispatch.py 全文级）

```python
_LEVEL_RANK = {"info": 0, "warn": 1, "critical": 2}
_REASON_TOKENS = {  # 稳定 ASCII 枚举（A3-F1/B3-5:禁 str(e) 入 dispatch——SMTP 应答含收件人/阿里云错误含手机号
  "throttled", "quota", "disabled", "timeout", "smtp_refused", "smtp_error",
  "enqueue", "submit", "im_partial", "no_binding", "not_configured", "expired",
}  # 原文异常只进 journald（logger=alert_dispatch）

_executor: ThreadPoolExecutor(max_workers=1, thread_name_prefix="alert-dispatch")  # 进程内惰性单例,daemon,随进程生灭——零新服务
# 队列有意 unbounded（F8:maxsize+丢弃=静默丢告警破审计承诺;风暴积压=延迟非丢失）

_producer: Celery("quant", broker=CELERY_BROKER_URL)  # B3-6:模块级惰性单例（仅 executor 单线程用,免锁）;
  # broker_transport_options={"socket_connect_timeout": 3, "socket_timeout": 5} + send_task(..., retry=False)
  # →broker 故障秒级快失败,直落降级直发（D-F1）;零 import scheduler.app/tasks（B2-P7）

def dispatch(level, category, title, body, code, notif_id):
    """notify() 惰性调用。submit 失败→log error+回写 _chain=failed:submit（仍不 raise）。
    notif_id=None（站内 insert 失败）→外推照跑（A3-F5:站内失败恰是外推价值时刻）,全线回写跳过。"""

def _dispatch_async(...):
    rows = _load_channels()            # DB 异常=空+warn
    if 无 enabled 行:
        if should_push_external(category, level):     # 过渡兜底（部署日不黑洞;订阅配好自然失效,下版本移除）
            ok = _push_channel(level, title, body, code, notif_id=notif_id)
            _writeback(notif_id, "legacy", "ok" if ok else "failed:timeout")
        else: _writeback_final_empty(notif_id)         # {} 空对象=链跑完零外推（B3-2 null 三义拆分）
        return
    matched = [row for row in rows if category∈row.categories and rank≥rank(row.min_level)]
    if not matched: _writeback_final_empty(notif_id); return
    for row in matched:
        if not SET NX alert:throttle:{ch}:{md5(title)} 1 EX 900: _writeback(ch,"skip:throttled"); continue
        n = INCR alert:quota:{ch}:{ZoneInfo("Asia/Shanghai"):%Y%m%d}; if n==1: EXPIRE 86400
        if n > LIMITS[ch]: _writeback(ch,"skip:quota"); continue
        payload = {row 快照, level, category, title/body 截断, code, notif_id}
        _writeback(notif_id, ch, "queued")            # A3-F3:先写 queued 再投（B3-1②守卫同挂,双保险）
        try: _producer.send_task(f"alerts.send_{ch}", queue=f"alerts_{ch}", expires=3600)
        except: _direct_send(row, ...)                # broker 故障降级:executor 线程内直发（D-F1;节流键已打不双发）
    # 配额 INCR 在 enqueue 侧一次计数;worker autoretry 不重复计数

def _writeback(notif_id, ch, value):
    """回写契约（A3-F2/F3/F4,B3-1/10 全收口）：
    ① notif_id None → return
    ② 单语句原子合并,禁 read-modify-write:
       UPDATE notifications SET dispatch = COALESCE(dispatch,'{}'::jsonb) || jsonb_build_object(%(ch)s,%(v)s)
       WHERE id=%(id)s [AND COALESCE(dispatch->>%(ch)s,'') !~ '^(ok|failed:|skip:)'  ← 仅 queued 写挂终态守卫]
    ③ 守卫 rowcount=0 → 弃写即安全（迟到的 queued 永不覆盖 worker 已落终态）
    ④ ok/failed/skip 终态写无条件
    ⑤ 自身失败 → logger.error("alert_dispatch writeback failed nid=%s ch=%s")——审计失败必须可见
    ⑥ reason 只允许 _REASON_TOKENS 枚举（failed:<token>）"""
```

**dispatch jsonb 值域（终态机）**：`{ch: ok|queued|failed:<token>|skip:<token>}` + `legacy`（过渡兜底）+ `_chain=failed:submit`（executor 入队失败）+ **`{}`=链跑完零外推**（终态）+ **`null`=未跑完/死亡窗**（唯一模糊态,网页渲染中性"未明"）。info/report 行永不写 dispatch（by design,前端不渲染）。
