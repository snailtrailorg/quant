## 接口契约

```python
# quant_common/domain/order.py
class Action(StrEnum): BUY = "BUY"; SELL = "SELL"
class OrderType(StrEnum): LIMIT = "limit"; MARKET = "market"
class Direction(StrEnum): LONG = "direction_long"; SHORT = "direction_short"; NET = "direction_net"
class OrderStatus(StrEnum):
    # 取值与 order_log 词表逐字一致：submitted/send_failed（提交侧）、
    # partial/all_traded/cancelled/rejected（交易所回报）、not_traded（仅表达，不落库）
    SUBMITTED="submitted"; SEND_FAILED="send_failed"; NOT_TRADED="not_traded"
    PARTIAL="partial"; ALL_TRADED="all_traded"; CANCELLED="cancelled"; REJECTED="rejected"
# ↑ 枚举值与字段名与现有 adapters.py 逐字一致（形状类以代码为准，D32 §12.2 判据）

@dataclass(frozen=True)
class Order:
    symbol: str; action: Action
    volume: float = 0; price: float = 0.0
    order_type: OrderType = OrderType.LIMIT
    client_id: str = ""                # 编排层生成 t{tid}:e{epoch}:c{n}
    # 无 offset / reference：A 股现货恒 Offset.NONE，reference 在前缀里（D32 §12.2）

@dataclass(frozen=True)
class OrderReceipt:
    client_id: str; accepted: bool; reason: str = ""

@dataclass(frozen=True)
class OrderState:
    orderid: str; vt_orderid: str; symbol: str; direction: Direction
    status: OrderStatus; volume: float; traded: float; price: float
    datetime: datetime | None = None

# quant_common/domain/trade.py
@dataclass(frozen=True)
class Trade:
    tradeid: str; orderid: str; symbol: str; direction: Direction
    price: float; volume: float; datetime: datetime | None = None

@dataclass(frozen=True)
class Position:
    symbol: str; volume: int; avg_price: float; pnl: float = 0.0
    direction: Direction = Direction.LONG
    frozen: int = 0; yd_volume: int = 0
    @property
    def available(self) -> int: ...    # max(volume - frozen, 0)，不存字段（单一算法）

# quant_common/domain/market.py
@dataclass(frozen=True)
class Tick:
    symbol: str; exchange: str; datetime: datetime   # tz-aware Asia/Shanghai
    last_price: float; volume: float = 0; turnover: float = 0
    open_price: float = 0; high_price: float = 0; low_price: float = 0
    pre_close: float = 0; limit_up: float = 0; limit_down: float = 0
    bid_price: tuple[float, ...]; ask_price: tuple[float, ...]      # 五档
    bid_volume: tuple[float, ...]; ask_volume: tuple[float, ...]

@dataclass(frozen=True)
class Bar:
    symbol: str; freq: str; ts: datetime             # UTC aware（BAR_COLUMNS 11 字段）
    open: float; high: float; low: float; close: float
    volume: float; amount: float
    adj_factor: float | None = None; source: str = ""

# quant_common/domain/account.py
@dataclass(frozen=True)
class AccountState:
    accountid: str; balance: float = 0.0; available: float = 0.0
    frozen: float = 0.0; margin: float = 0.0

@dataclass(frozen=True)
class Health:
    ok: bool; state: str = "ok"; detail: str = ""
```

# ports/execution.py
class ExecutionPort(Protocol):
    def connect(self) -> None: ...
    def close(self) -> None: ...

    def connected(self) -> bool: ...
    def send(self, order: Order) -> OrderReceipt: ...
    def cancel(self, client_order_id: str) -> None: ...
    def subscribe_order(self, cb: Callable[[OrderState], None]) -> None: ...
    def subscribe_trade(self, cb: Callable[[Trade], None]) -> None: ...
    def query_positions(self) -> list[Position]: ...
    def query_account(self) -> AccountState | None: ...
    def query_orders(self) -> list[OrderState]: ...
    def query_trades(self) -> list[Trade]: ...
    def resolve_vt_order_id(self, client_order_id: str) -> str | None: ...

# ports/market_data.py
class MarketDataPort(Protocol):
    def connect(self, credentials: dict, params: dict) -> None: ...
    def close(self) -> None: ...
    @property
    def connected(self) -> bool: ...
    def start_ready(self) -> bool: ...
    def set_context(self, ctx: dict) -> None: ...
    def poll_supervise(self, trading_day: bool) -> None: ...
    def event_threads_healthy(self) -> bool: ...
    def set_on_tick(self, cb: Callable[[Tick], None]) -> None: ...
    def subscribe(self, symbols: list[str]) -> None: ...
    def unsubscribe(self, symbols: list[str]) -> None: ...
    def snapshot(self, symbol: str) -> Tick | None: ...
