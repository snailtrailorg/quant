## 接口契约

```python
# quant_common/domain/order.py
class Side(StrEnum): BUY = "BUY"; SELL = "SELL"
class Offset(StrEnum): NONE = "NONE"; OPEN = "OPEN"; CLOSE = "CLOSE"; CLOSE_TODAY = "CLOSE_TODAY"
class OrderStatus(StrEnum): SUBMITTING; NOT_TRADED; PART_TRADED; ALL_TRADED; CANCELLED; REJECTED

@dataclass(frozen=True)
class Order:
    symbol: str; exchange: str; direction: Side; offset: Offset
    order_type: str; volume: float; price: float
    client_order_id: str = ""          # 编排层生成 t{tid}:e{epoch}:c{n}
    reference: str = ""

@dataclass(frozen=True)
class OrderReceipt:
    client_order_id: str; accepted: bool; reason: str = ""

# quant_common/domain/position.py
@dataclass(frozen=True)
class Position:
    symbol: str; exchange: str; direction: Side
    volume: float; frozen: float; yd_volume: float; available: float
    cost_price: float; pnl: float

# quant_common/domain/tick.py / bar.py
@dataclass(frozen=True)
class Tick:
    symbol: str; ts: datetime; last: float; bid: float; ask: float
    bid_vol: float; ask_vol: float; lower_limit: float | None; upper_limit: float | None
@dataclass(frozen=True)
class Bar:
    symbol: str; ts: datetime; open: float; high: float; low: float; close: float
    volume: float; turnover: float

# quant_common/domain/account.py
@dataclass(frozen=True)
class AccountState:
    account_id: int; available_cash: float; margin: float; balance: float; frozen_cash: float

@dataclass(frozen=True)
class Health:
    ok: bool; state: str; detail: str = ""
```

```python
# ports/execution.py
class ExecutionPort(Protocol):
    def connect(self) -> None: ...
    def close(self) -> None: ...
    @property
    def connected(self) -> bool: ...
    def send(self, order: Order) -> OrderReceipt: ...
    def cancel(self, client_order_id: str) -> None: ...
    def subscribe_order(self, cb: Callable[[OrderStatus], None]) -> None: ...
    def subscribe_trade(self, cb: Callable[[Any], None]) -> None: ...
    def query_positions(self) -> list[Position]: ...
    def query_account(self) -> AccountState: ...
    def query_orders(self) -> list[Any]: ...
    def query_trades(self) -> list[Any]: ...
    def resolve_vt_order_id(self, client_order_id: str) -> str | None: ...

# ports/market_data.py
class MarketDataPort(Protocol):
    def connect(self, credentials: dict, params: dict) -> None: ...
    def close(self) -> None: ...
    @property
    def connected(self) -> bool: ...
    def start_ready(self) -> bool: ...
    def set_context(self, ctx: dict) -> None: ...
    def poll_supervise(self, trading_day: bool) -> None: ...
    def event_threads_healthy(self) -> bool: ...
    def set_on_tick(self, cb: Callable[[Tick], None]) -> None: ...
    def subscribe(self, symbols: list[str]) -> None: ...
    def unsubscribe(self, symbols: list[str]) -> None: ...
    def snapshot(self, symbol: str) -> Tick | None: ...
```
