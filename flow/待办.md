```sql
-- 行数(按符号,兼看两侧订阅面)
SELECT 'hub' t, symbol, count(*) FROM bar_hub WHERE ts::date=current_date GROUP BY symbol
UNION ALL SELECT 'shadow', symbol, count(*) FROM bar_shadow WHERE ts::date=current_date GROUP BY symbol;
-- 逐根差异(OHLCV+amount,14 号 §113)
SELECT h.symbol, h.ts FROM bar_hub h JOIN bar_shadow s USING (symbol, ts)
WHERE h.ts::date=current_date
  AND (h.open<>s.open OR h.high<>s.high OR h.low<>s.low OR h.close<>s.close
       OR h.volume<>s.volume OR h.amount<>s.amount);
-- anti-join(JOIN 查不出缺根;仅限 shadow 已覆盖符号,08-28 实证 112 vs 117;
--   豁免:①进程首根 shadow 缺=hub_only 一根(以 worker 日志「shadow 首根跳过」为准,不按时刻——盘中崩溃重启同理,P2 修复批 08-28 起)②每日 11:31 shadow_only 一根/符号(hub 修后 11:30:0x 被滤不产,vnpy 无时段过滤照产)③部署重启日:worker 首 tick 前整段 hub_only(08-28 实证 worker 首笔延迟 11 分钟,以「shadow 首根跳过」日志时刻为界)
SELECT 'hub_only' side, h.symbol, h.ts FROM bar_hub h LEFT JOIN bar_shadow s USING (symbol, ts)
WHERE h.ts::date=current_date AND s.symbol IS NULL
  AND h.symbol IN (SELECT DISTINCT symbol FROM bar_shadow WHERE ts::date=current_date)
UNION ALL
SELECT 'shadow_only', s.symbol, s.ts FROM bar_shadow s LEFT JOIN bar_hub h USING (symbol, ts)
WHERE s.ts::date=current_date AND h.symbol IS NULL;
```
